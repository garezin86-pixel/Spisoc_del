# ============================================================
# ПОСЛЕ — сервис приходит снаружи через Depends
# ============================================================

from fastapi import APIRouter, Depends
from src.dependencies import get_user_service, get_task_service
from src.services.user_service import UserService
from src.services.task_service import TaskService
from src.models.user import UserModel
# from src.core.security import get_current_user  # ваш текущий dep

router = APIRouter()


@router.get("/users/{user_id}")
async def get_user(
    user_id: int,
    service: UserService = Depends(get_user_service),
    # current_user: UserModel = Depends(get_current_user),
):
    return await service.get_user(user_id, current_user=..., )  # type: ignore


@router.get("/tasks/")
async def get_tasks(
    service: TaskService = Depends(get_task_service),
    # current_user: UserModel = Depends(get_current_user),
):
    return await service.filter_tasks(current_user=..., filter_user_group=None,
                                      group_id=None, filter_type=None,
                                      is_done=None, limit=20, offset=0)
